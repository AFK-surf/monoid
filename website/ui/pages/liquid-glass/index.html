MoonBit API & Attribution

Real @sf and @glass APIs, straight from production.

// libs/ui/src/studio_pages/studio.mbt
let range = @signals.signal(0)
let refract = @signals.signal(true)
let jobs = @signals.signal(8.0)

@sf.button_node(row, fn() { "Run Build" }, fn() { /* action */ }, tone=Primary) |> ignore

@sf.switch_node(row, fn() { refract.get() }, fn(v) { refract.set(v) }, fn() { "Bezel Specular" })

@sf.seg_node(row, fn() { ["Stable", "Beta", "Canary"] }, fn() { range.get() }, fn(i) { range.set(i) }) |> ignore

@sf.value_slider_node(col, fn() { "Parallel Jobs" }, fn() { jobs.get() }, fn(v) { jobs.set(v) }, min=1.0, max=16.0) |> ignore

// Draggable Liquid Glass segmented control (@glass)
@glass.liquid_segmented_node(
  col,
  fn() { ["Day", "Week", "Month"] },
  fn() { range.get() },
  fn(i) { range.set(i) },
  glass_track=true,
) |> ignore

// 360-degree capillary liquid bridge (@motion.LiquidBond)
let bond = @glass.LiquidMerge::new(separated=false)
@glass.liquid_merge_node(
  col,
  controller=bond,
  main_label=fn() { "Capsule" },
) |> ignore
