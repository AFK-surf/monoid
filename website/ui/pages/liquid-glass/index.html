Compose glass in plain MoonBit.

// libs/liquid-glass/src/pages/pages.mbt
let wifi = @signals.signal(true)
let brightness = @signals.signal(40.0)
let tab = @signals.signal(0)

@lg.liquid_button_node(
  row,
  fn() { "Tinted Liquid Button" },
  fn() { /* action */ },
  tint=0x0088FFFFL,
) |> ignore

@lg.liquid_toggle_node(
  row,
  fn() { wifi.get() },
  fn(v) { wifi.set(v) },
  fn() { "Wi-Fi" },
) |> ignore

@lg.liquid_slider_node(
  col,
  fn() { brightness.get() },
  fn(v) { brightness.set(v) },
  fn() { "Brightness" },
) |> ignore

@lg.liquid_bottom_tabs_node(
  col,
  ["Optics", "Deck", "Loupe", "Merge"],
  fn() { tab.get() },
  fn(i) { tab.set(i) },
  label="Tabs",
) |> ignore

// iOS 27 capillary liquid bridge (@motion.LiquidBond)
let bond = @lg.LiquidMerge::new(separated=false)
@lg.liquid_merge_node(
  col,
  controller=bond,
  main_label=fn() { "Liquid Capsule" },
) |> ignore

Glass::liquid unifies Apple iOS 27 UIGlassEffect optics (uniform blur, quarter-circle bevel refraction, capillary liquid bridge hysteresis via @motion.LiquidBond and @lg.liquid_merge_node, ios27Light / ios27Dark / ios27Clear, inner bevel shadow & outline, touch spot, and .sharing() capture) based on whynotmake-it/flutter_liquid_glass, with the interactive control catalog adapted from Kyant0/AndroidLiquidGlass.
